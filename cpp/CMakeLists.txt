cmake_minimum_required(VERSION 3.18)
project(tapnext_demo CXX)
set(CMAKE_CXX_STANDARD 17)
# -DONNXRUNTIME_ROOT=/path/to/onnxruntime-linux-x64-<ver> (release tarball)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime install dir")
option(WITH_TENSORRT "Build the TensorRT backend (tapnext_trt_tracker.hpp)" OFF)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT install dir (include/, lib/)")

find_package(OpenCV REQUIRED core imgproc videoio imgcodecs)
add_executable(tapnext_demo demo.cpp)
target_include_directories(tapnext_demo PRIVATE ${ONNXRUNTIME_ROOT}/include)
target_link_directories(tapnext_demo PRIVATE ${ONNXRUNTIME_ROOT}/lib)
target_link_libraries(tapnext_demo PRIVATE onnxruntime ${OpenCV_LIBS})
set_target_properties(tapnext_demo PROPERTIES BUILD_RPATH ${ONNXRUNTIME_ROOT}/lib)

if(WITH_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(tapnext_demo PRIVATE TAPNEXT_WITH_TRT)
  target_include_directories(tapnext_demo PRIVATE ${TENSORRT_ROOT}/include)
  target_link_directories(tapnext_demo PRIVATE ${TENSORRT_ROOT}/lib)
  target_link_libraries(tapnext_demo PRIVATE nvinfer CUDA::cudart)
endif()
