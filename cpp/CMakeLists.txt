cmake_minimum_required(VERSION 3.18)
project(trackon_trt_demo CXX)
set(CMAKE_CXX_STANDARD 17)
# -DTENSORRT_ROOT=<TensorRT install, e.g. D:/SVN/PTnD/extern/tensorrt_10.14.1.48_CUDA13>
set(TENSORRT_ROOT "" CACHE PATH "TensorRT install dir (include/, lib/)")
find_package(CUDAToolkit REQUIRED)
find_package(OpenCV REQUIRED core imgproc videoio imgcodecs)
add_executable(trackon_demo demo.cpp)
target_include_directories(trackon_demo PRIVATE ${TENSORRT_ROOT}/include)
target_link_directories(trackon_demo PRIVATE ${TENSORRT_ROOT}/lib)
if(WIN32)
  # TensorRT 10 on Windows names the import library with the major version
  file(GLOB NVINFER_LIB "${TENSORRT_ROOT}/lib/nvinfer_10.lib" "${TENSORRT_ROOT}/lib/nvinfer.lib")
  list(GET NVINFER_LIB 0 NVINFER_LIB)
  target_link_libraries(trackon_demo PRIVATE ${NVINFER_LIB} CUDA::cudart ${OpenCV_LIBS})
else()
  target_link_libraries(trackon_demo PRIVATE nvinfer CUDA::cudart ${OpenCV_LIBS})
endif()
